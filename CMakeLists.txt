cmake_minimum_required(VERSION 3.20)
project(makemarketbench LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()

add_library(mmb_core STATIC cpp/src/order_book.cpp)
target_include_directories(mmb_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/cpp/include)
set_target_properties(mmb_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(NOT MSVC)
  target_compile_options(mmb_core PRIVATE -Wall -Wextra -Wpedantic)
endif()

option(MMB_BUILD_NATIVE "Build native C++ tests and benchmark" OFF)
if(MMB_BUILD_NATIVE)
  add_executable(test_book cpp/tests/test_book.cpp)
  target_link_libraries(test_book PRIVATE mmb_core)
  add_executable(bench_book cpp/bench/bench_book.cpp)
  target_link_libraries(bench_book PRIVATE mmb_core)
  target_include_directories(bench_book PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/cpp/bench)
  enable_testing()
  add_test(NAME test_book COMMAND test_book)
endif()

if(SKBUILD)
  find_package(pybind11 CONFIG REQUIRED)
  pybind11_add_module(_engine cpp/src/bindings.cpp)
  target_link_libraries(_engine PRIVATE mmb_core)
  install(TARGETS _engine DESTINATION makemarket)
endif()
